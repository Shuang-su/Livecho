from collections import Counter
from pathlib import Path
import json
import re

import unicodedata

BASE = "93d8b0a4b3e147ce0b7df903f118060117d56f0a"
ROOT = Path.cwd()
DELIVERABLES = [Path(p) for p in (
    "README.md", "docs/roadmap.md", "docs/requirements-index.md",
    "docs/changes/30-requirements-evidence-index/evidence.md",
)]
SOURCES = [Path("docs") / (p + ".md") for p in (
    "architecture/adr/0001-alpha-modular-monolith",
    "security/alpha-threat-model", "security/data-lifecycle-and-deletion",
    "policy/bilibili-public-ingest", "policy/independent-implementation",
    "operations/incident-disable-and-recovery",
    "operations/railway-deployment", "operations/railway-secrets",
)]
PATTERN = r"\b(?:DEC|FLOW|CTRL|DATA|THREAT|RISK|BILI|LIC|GATE)-[A-Z0-9]+(?:-[A-Z0-9]+)*"

def prose(path):
    return re.sub(r"```.*?```", "", path.read_text(), flags=re.S)

def anchors(path):
    result, counts = set(re.findall(r'<a id="([^"]+)"', path.read_text())), Counter()
    for heading in re.findall(r"^#{1,6} (.+)$", prose(path), re.M):
        slug = "".join(c for c in heading.lower() if c in " -_" or unicodedata.category(c)[0] in "LN")
        slug = slug.replace(" ", "-")
        count = counts[slug]
        counts[slug] += 1
        result.add(slug + (f"-{count}" if count else ""))
    return result

local_links, external_links = 0, set()
for path in DELIVERABLES:
    text = prose(path)
    refs = dict(re.findall(r"^\[([^\]]+)\]:\s*(\S+)", text, re.M))
    targets = re.findall(r"\[[^\]\n]+\]\(([^)\n]+)\)", text)
    for label in re.findall(r"\[[^\]\n]+\]\[([^\]\n]+)\]", text):
        assert label in refs, (path, "undefined reference", label)
        targets.append(refs[label])
    targets.extend(refs.values())
    for target in targets:
        if re.match(r"https?://", target):
            external_links.add(target)
            continue
        file_part, _, fragment = target.partition("#")
        dest = (path.parent / file_part).resolve() if file_part else path.resolve()
        assert dest.exists(), (path, "missing file", target)
        if fragment:
            assert dest.is_file() and fragment in anchors(dest), (path, "missing anchor", target)
        local_links += 1

source_ids = set()
for path in SOURCES:
    source_ids.update(re.findall(PATTERN, path.read_text()))
source_ids -= {"BILI-SRC", "BILI-RIGHT"}  # Wildcard prefixes, not definition IDs.
index = Path("docs/requirements-index.md").read_text()
records = re.findall(r"^\| ((?:IDX|DEC|FLOW|CTRL|DATA|THREAT|RISK|BILI|LIC|GATE)-[A-Z0-9-]+) — (.*)$", index, re.M)
counts = Counter(key for key, _ in records)
assert all(count == 1 for count in counts.values()), counts
indexed_stable = {key for key in counts if not key.startswith("IDX-")}
assert source_ids == indexed_stable, {"missing": sorted(source_ids-indexed_stable), "extra": sorted(indexed_stable-source_ids)}
for key, rest in records:
    cells = rest.split("|")
    assert len(cells) == 5 and all(c.strip() for c in cells[:4]), (key, cells)
    assert len(re.findall(r"\[[^\]]+\](?:\([^)]+\)|\[[^\]]+\])", cells[1])) == 1, key
    assert re.search(r"\]\(#(?:foundation|design|protocol|railway|policy|license|pending)\)", cells[3]), key

control_owners = {}
